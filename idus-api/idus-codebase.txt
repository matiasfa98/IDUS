// src/scripts/migrateMetaphors.js
//
// Migrates User.metaphor from the old flat shape
//   { key, label, animeTheme, customArtUrl }
// to the new two-layer shape
//   { genre, archetype, vibe, displayName, tagline, art }
//
// Run: node src/scripts/migrateMetaphors.js
// Idempotent. Backs off if `metaphor.genre` is already set.

import mongoose from "mongoose";
import { config } from "../src/config.js";
import { User } from "../src/models/User.js";
import { DEFAULT_METAPHOR } from "../src/data/metaphors.js";

async function main() {
  await mongoose.connect(config.mongoUri);
  console.log("[migrate] connected");

  const cursor = User.find({
    $or: [
      { "metaphor.genre": { $exists: false } },
      { "metaphor.genre": null },
    ],
  }).cursor();

  let count = 0;
  for await (const user of cursor) {
    const old = user.metaphor ?? {};
    user.metaphor = {
      genre: DEFAULT_METAPHOR.genre,
      archetype: DEFAULT_METAPHOR.archetype,
      vibe: DEFAULT_METAPHOR.vibe,
      displayName: old.label ?? null,
      tagline: old.animeTheme ?? null,
      art: old.customArtUrl
        ? {
            source: "upload",
            url: old.customArtUrl,
            thumbUrl: null,
            attribution: null,
            externalId: null,
          }
        : null,
      updatedAt: new Date(),
    };
    await user.save();
    count++;
  }

  console.log(`[migrate] done — ${count} user(s) updated`);
  await mongoose.disconnect();
}

main().catch((err) => {
  console.error("[migrate] failed", err);
  process.exit(1);
});

import { getAuth } from "@clerk/express";
import { User } from "../models/User.js";
import { generatePermanentId } from "../utils/permanentId.js";

export async function requireAuth(req, res, next) {
  try {
    const auth = getAuth(req);
    console.log("[auth] getAuth:", JSON.stringify(auth));
    console.log("[auth] header starts:", req.headers.authorization?.slice(0, 30));

    const { userId } = auth;

    if (!userId) {
      return res.status(401).json({ error: "UNAUTHENTICATED" });
    }

    let user = await User.findOne({ clerkId: userId });
    if (!user) {
      user = await User.create({
        clerkId: userId,
        permanentId: await generatePermanentId(),
      });
    }

    req.userId = String(user._id);
    req.clerkId = userId;
    next();
  } catch (err) {
    console.error("[auth] error:", err);
    next(err);
  }
}

// src/middleware/rateLimit.js
//
// Sliding-window rate limiter. In-memory only — single process.
// For multi-instance scale, swap the Map for Redis (INCR + EXPIRE).

const buckets = new Map(); // key -> array of timestamps (ms), ascending

// Periodic cleanup to avoid unbounded growth.
const CLEANUP_INTERVAL_MS = 60 * 1000;
setInterval(() => {
  const now = Date.now();
  for (const [key, timestamps] of buckets.entries()) {
    const pruned = timestamps.filter((t) => now - t < 60 * 60 * 1000);
    if (pruned.length === 0) buckets.delete(key);
    else buckets.set(key, pruned);
  }
}, CLEANUP_INTERVAL_MS).unref();

/**
 * Create a rate-limit middleware.
 *
 * @param {object} opts
 * @param {number} opts.windowMs - window duration in ms
 * @param {number} opts.max - max requests per window
 * @param {string} opts.name - identifier for error messages + key namespacing
 * @param {(req: import("express").Request) => string} [opts.keyFn]
 * @returns {import("express").RequestHandler}
 */
export function rateLimit({ windowMs, max, name, keyFn }) {
  return (req, res, next) => {
    const key = `${name}:${keyFn ? keyFn(req) : defaultKey(req)}`;
    const now = Date.now();
    const windowStart = now - windowMs;

    // Prune expired timestamps for this key.
    let timestamps = (buckets.get(key) ?? []).filter((t) => t > windowStart);

    const wouldExceed = timestamps.length >= max;

    if (wouldExceed) {
      // Compute when the user can retry.
      // We have N = timestamps.length requests in the window.
      // We need to get down to max requests.
      // The oldest (N - max + 1) timestamps must expire before we accept.
      // The one that matters is timestamps[N - max] (0-indexed).
      const indexToExpire = timestamps.length - max;
      const pivot = timestamps[indexToExpire] ?? timestamps[0] ?? now;
      const retryAt = pivot + windowMs;
      const retryAfterSec = Math.max(1, Math.ceil((retryAt - now) / 1000));

      // Do NOT push the current request into the bucket —
      // rejected requests shouldn't count toward the next window.
      buckets.set(key, timestamps);

      res.setHeader("X-RateLimit-Limit", String(max));
      res.setHeader("X-RateLimit-Remaining", "0");
      res.setHeader("X-RateLimit-Reset", String(Math.ceil(retryAt / 1000)));
      res.setHeader("Retry-After", String(retryAfterSec));

      return res.status(429).json({
        error: "RATE_LIMITED",
        message: `Too many requests. Try again in ${retryAfterSec}s.`,
        retryAfter: retryAfterSec,
      });
    }

    // Accept: push and continue.
    timestamps.push(now);
    buckets.set(key, timestamps);

    const remaining = Math.max(0, max - timestamps.length);
    // Reset time = when the oldest accepted request leaves the window.
    const oldest = timestamps[0] ?? now;
    const resetAt = oldest + windowMs;

    res.setHeader("X-RateLimit-Limit", String(max));
    res.setHeader("X-RateLimit-Remaining", String(remaining));
    res.setHeader("X-RateLimit-Reset", String(Math.ceil(resetAt / 1000)));

    next();
  };
}

function defaultKey(req) {
  if (req.userId) return `u:${req.userId}`;
  const forwarded = req.headers["x-forwarded-for"];
  if (typeof forwarded === "string") {
    return `ip:${forwarded.split(",")[0].trim()}`;
  }
  return `ip:${req.ip}`;
}

export function validate(schema, source = "body") {
  return (req, res, next) => {
    const result = schema.safeParse(req[source]);

    if (!result.success) {
      return res.status(400).json({
        error: "VALIDATION_ERROR",
        details: result.error.flatten(),
      });
    }

    req[source] = result.data;
    next();
  };
}

      type: mongoose.Schema.Types.ObjectId,

  {
    conversationId: {
      type: mongoose.Schema.Types.ObjectId,
      ref: "Conversation",
      required: true,
      index: true,
    },
    senderId: {
      type: mongoose.Schema.Types.ObjectId,
      ref: "User",
      required: true,
      index: true,
    },

    // ULID — sortable by creation time, generated by the client.
    // Also used for idempotency: unique per sender.
    clientMessageId: { type: String, required: true },

    body: { type: String, default: "" },
    attachments: { type: [attachmentSchema], default: [] },

    replyToId: {
      type: mongoose.Schema.Types.ObjectId,
      ref: "Message",
      default: null,
    },

    reactions: { type: mongoose.Schema.Types.Mixed, default: {} },

    editedAt: { type: Date, default: null },
    deletedAt: { type: Date, default: null },

    // Client's clock, for offline ordering. Server's createdAt is authoritative.
    localTimestamp: { type: Date, required: true },
  },
  { timestamps: true }
);

// Idempotency: same sender cannot insert the same clientMessageId twice.
messageSchema.index(
  { senderId: 1, clientMessageId: 1 },
  { unique: true }
);

// History pagination — newest first per conversation.
messageSchema.index({ conversationId: 1, createdAt: -1 });

// Reply lookup (rare).
messageSchema.index({ replyToId: 1 }, { sparse: true });

export const Message = mongoose.model("Message", messageSchema);

// src/routes/attachments.js
import express from "express";
import { z } from "zod";

import { requireAuth } from "../middleware/auth.js";
import { validate } from "../middleware/validate.js";
import {
  buildSignedUpload,
  fetchResourceMetadata,
  isMimeAllowed,
  SIZE_LIMITS,
} from "../services/cloudinary.js";
import { uploadSignLimiter } from "../middleware/rateLimit.js";

export const attachmentsRouter = express.Router();

// ── Schemas ──────────────────────────────────────────
const signSchema = z.object({
  kind: z.enum(["image", "file"]),
  size: z.number().int().positive(),
  mimeType: z.string().min(3).max(120),
  originalName: z.string().max(255).optional(),
});

const confirmSchema = z.object({
  kind: z.enum(["image", "file"]),
  publicId: z.string().min(3).max(300),
});

// ── POST /attachments/sign ───────────────────────────
// Client asks: "I want to upload an image/file of X bytes with Y mime type."
// Server validates, then returns signed upload params.
attachmentsRouter.post(
  "/sign",
  requireAuth,
  uploadSignLimiter,
  validate(signSchema),
  async (req, res, next) => {
    try {
      const { kind, size, mimeType } = req.body;

      // Size check
      const limit = SIZE_LIMITS[kind];
      if (!limit) {
        return res.status(400).json({
          error: "UNSUPPORTED_KIND",
          message: `kind must be "image" or "file"`,
        });
      }
      if (size > limit) {
        return res.status(413).json({
          error: "FILE_TOO_LARGE",
          message: `Max ${limit} bytes for ${kind}. Received ${size}.`,
          maxBytes: limit,
        });
      }

      // MIME allowlist
      if (!isMimeAllowed(kind, mimeType)) {
        return res.status(415).json({
          error: "UNSUPPORTED_MIME",
          message: `MIME type ${mimeType} is not allowed for ${kind}.`,
        });
      }

      const signed = buildSignedUpload({
        kind,
        userId: String(req.userId),
      });

      res.json({
        uploadUrl: signed.uploadUrl,
        resourceType: signed.resourceType,
        params: signed.params,
        // Client should read these to know the caps
        limits: { maxBytes: limit },
      });
    } catch (err) {
      next(err);
    }
  }
);

// ── POST /attachments/confirm ────────────────────────
// After Cloudinary upload succeeds, the client calls this to prove the
// publicId exists and get back canonical URLs + metadata.
attachmentsRouter.post(
  "/confirm",
  requireAuth,
  validate(confirmSchema),
  async (req, res, next) => {
    try {
      const { kind, publicId } = req.body;

      // Ownership check: the publicId must live under our per-user folder.
      const expectedPrefix = `attachments/${
        kind === "image" ? "image" : "raw"
      }/${req.userId}/`;
      if (!publicId.startsWith(expectedPrefix)) {
        return res.status(403).json({
          error: "NOT_YOUR_RESOURCE",
          message: `publicId must start with "${expectedPrefix}".`,
        });
      }

      const meta = await fetchResourceMetadata(publicId, kind);
      if (!meta) {
        return res.status(404).json({
          error: "RESOURCE_NOT_FOUND",
          message: "Cloudinary has no such resource in your account.",
        });
      }

      res.json({
        attachment: {
          kind,
          url: meta.url,
          thumbUrl: meta.thumbUrl,
          mimeType: meta.mimeType,
          size: meta.size,
          width: meta.width,
          height: meta.height,
          cloudinaryPublicId: publicId,
        },
      });
    } catch (err) {
      next(err);
    }
  }
);

async function upsertCreatorParticipant(challengeId, userId) {
  return Participant.updateOne(
    { challengeId, userId },
    {

        challengeId,

        role: "MODERATOR",
        joinedAt: new Date(),

/**
 * Ensure the challenge has a conversation. Called on publish.
 * Idempotent — safe to call repeatedly.
 */
async function ensureChallengeConversation(challenge) {
  let convo = await Conversation.findOne({ challengeId: challenge._id });

  if (!convo) {
    convo = await Conversation.create({
      type: "challenge",
      challengeId: challenge._id,
      participants: [
        {
          userId: challenge.creatorId,
          role: "ADMIN",
          joinedAt: new Date(),
        },
      ],
      createdBy: challenge.creatorId,
    });
    console.log(
      `[challenges] created conversation ${convo._id} for ${challenge.slug}`
    );
  }

  return convo;
}

/**
 * Add a user to the challenge conversation if not already an active participant.
 * Idempotent.
 */
async function addConversationParticipant(challengeId, userId, role) {
  const convo = await Conversation.findOne({ challengeId });
  if (!convo) return null;

  const alreadyIn = convo.participants.some(
    (p) => String(p.userId) === String(userId) && !p.leftAt
  );
  if (alreadyIn) return convo;

  convo.participants.push({
    userId,
    role,
    joinedAt: new Date(),
  });
  await convo.save();
  return convo;
}

/**
 * Mark a user as having left the challenge conversation.
 * Idempotent.
 */
async function removeConversationParticipant(challengeId, userId) {
  const convo = await Conversation.findOne({ challengeId });
  if (!convo) return null;

  const p = convo.participants.find(
    (x) => String(x.userId) === String(userId) && !x.leftAt
  );
  if (!p) return convo;

  p.leftAt = new Date();
  await convo.save();
  return convo;
}

      const scheduledDates = resolveChallengeDays(
        challenge.startDate,
        challenge.endDate,
        challenge.schedule
      );

      const phases = progression.phases.map((phase, phaseIdx) => ({
        order: phaseIdx,
        title: phase.title,
        description: phase.description,
        monthLabel: phase.monthLabel,
        days: phase.days.map((day) => {
          const scheduled = scheduledDates[day.dayNumber - 1];
          return {
            _id: new mongoose.Types.ObjectId(),
            dayNumber: day.dayNumber,
            scheduledDate: scheduled,
            title: day.title,
            type: day.type,
            description: day.description,
            objective: day.objective,
            requirements: day.requirements,
            estimatedMinutes: day.estimatedMinutes,
            isFinal: day.type === "FINAL",
          };
        }),
      }));

      challenge.phases = phases;
      challenge.version += 1;
      await challenge.save();

      res.json({
        challenge: serializeChallenge(challenge),
        generated: {
          phaseCount: phases.length,
          dayCount: progression.phases.reduce((n, p) => n + p.days.length, 0),
          summary: progression.challenge.summary,
        },
      });
    } catch (err) {
      next(err);
    }
  }
);

challengesRouter.post(
  "/:slug/publish",
  requireAuth,
  async (req, res, next) => {
    try {
      const challenge = await Challenge.findOne({ slug: req.params.slug });
      if (!challenge) {
        return res.status(404).json({ error: "CHALLENGE_NOT_FOUND" });
      }
      if (String(challenge.creatorId) !== String(req.userId)) {
        return res.status(403).json({ error: "FORBIDDEN" });
      }

      const alreadyPublished = challenge.status === "PUBLISHED";

      if (!alreadyPublished) {
        if (!challenge.phases || challenge.phases.length === 0) {
          return res.status(409).json({
            error: "NO_PHASES",
            message:
              "Generate phases first: POST /challenges/:slug/generate",
          });
        }

        // Upsert participant BEFORE flipping status. If this throws,
        // the challenge stays DRAFT and /publish can be retried.
        await upsertCreatorParticipant(challenge._id, req.userId);

        challenge.status = "PUBLISHED";
        challenge.version += 1;
        await challenge.save();
      } else {
        // Already published: still make sure the creator is a participant.
        await upsertCreatorParticipant(challenge._id, req.userId);
      }

      // Ensure the challenge conversation exists (idempotent).
      await ensureChallengeConversation(challenge);

      res.json({
        challenge: serializeChallenge(challenge),
        alreadyPublished,
      });
    } catch (err) {
      next(err);
    }
  }
);

function dmKeyFor(a, b) {

function serializeParticipant(p, user) {
  return {
    userId: String(p.userId),
    role: p.role,
    joinedAt: p.joinedAt,
    leftAt: p.leftAt,
    lastReadMessageId: p.lastReadMessageId ? String(p.lastReadMessageId) : null,

  };
}

async function serializeConversation(convo, { includeParticipants = true } = {}) {
  const base = {
    id: String(convo._id),
    type: convo.type,
    challengeId: convo.challengeId ? String(convo.challengeId) : null,
    title: convo.title,
    avatarUrl: convo.avatarUrl,
    createdBy: String(convo.createdBy),
    lastMessageAt: convo.lastMessageAt,
    lastMessagePreview: convo.lastMessagePreview,
    lastMessageSenderId: convo.lastMessageSenderId
      ? String(convo.lastMessageSenderId)

    messageCount: convo.messageCount ?? 0,

  const userMap = new Map(users.map((u) => [String(u._id), u]));

  return {
    ...base,
    participants: convo.participants.map((p) =>
      serializeParticipant(p, userMap.get(String(p.userId)))
    ),
  };
}

// Create a DM or group. Idempotent for DMs (via dmKey).
conversationsRouter.post(
  "/",
  requireAuth,
  validate(createConversationSchema),
  async (req, res, next) => {
    try {
      const { type, participantIds, title } = req.body;

      // Never include self twice.
      const cleanIds = Array.from(
        new Set(participantIds.filter((id) => String(id) !== String(req.userId)))
      );

      if (cleanIds.length === 0) {
        return res.status(400).json({
          error: "NO_OTHER_PARTICIPANTS",
          message: "Provide at least one other participant.",
        });
      }

      // Verify all participants exist.
      const users = await User.find({ _id: { $in: cleanIds } }, { _id: 1 }).lean();
      if (users.length !== cleanIds.length) {
        return res.status(400).json({
          error: "UNKNOWN_PARTICIPANT",
          message: "One or more participantIds do not exist.",
        });
      }

      // ── DM: exactly 2 people, idempotent via dmKey ─
      if (type === "dm") {
        if (cleanIds.length !== 1) {
          return res.status(400).json({
            error: "DM_REQUIRES_ONE_PARTICIPANT",
            message: "A DM is between exactly two users.",
          });
        }

        const otherId = cleanIds[0];
        const dmKey = dmKeyFor(req.userId, otherId);

        let convo = await Conversation.findOne({ dmKey });

        if (!convo) {
          convo = await Conversation.create({
            type: "dm",
            dmKey,
            participants: [
              { userId: req.userId, role: "ADMIN", joinedAt: new Date() },
              { userId: otherId, role: "MEMBER", joinedAt: new Date() },
            ],
            createdBy: req.userId,
          });
        }

        return res.status(200).json({
          conversation: await serializeConversation(convo),
          alreadyExisted:
            !!convo.createdAt && convo.createdAt < new Date(Date.now() - 500),
        });
      }

      // ── Group: 3+ people, no idempotency key ────────
      const participants = [
        { userId: req.userId, role: "ADMIN", joinedAt: new Date() },
        ...cleanIds.map((id) => ({
          userId: id,
          role: "MEMBER",
          joinedAt: new Date(),
        })),
      ];

      const convo = await Conversation.create({
        type: "group",
        title: title ?? null,
        participants,
        createdBy: req.userId,
      });

      res.status(201).json({
        conversation: await serializeConversation(convo),
      });
    } catch (err) {
      next(err);
    }
  }
);

        Authorization: `Bearer ${config.clerk.secretKey}`,
        "Content-Type": "application/json",
      },
      body: JSON.stringify({ user_id: userId }),
    });

    if (!sessionRes.ok) {
      const text = await sessionRes.text();
      console.error("[dev/token] session create failed:", sessionRes.status, text);
      return res.status(sessionRes.status).json({
        error: "CLERK_SESSION_CREATE_FAILED",
        message: text,
      });
    }

    const session = await sessionRes.json();

    // ── Mint token ─────────────────────────────────
    const tokenRes = await fetch(
      `https://api.clerk.com/v1/sessions/${session.id}/tokens?expires_in_seconds=${REQUESTED_TTL_SECONDS}`,
      {
        method: "POST",
        headers: {
          Authorization: `Bearer ${config.clerk.secretKey}`,
          "Content-Type": "application/json",
        },
      }
    );

    if (!tokenRes.ok) {
      const text = await tokenRes.text();
      console.error("[dev/token] token mint failed:", tokenRes.status, text);
      return res.status(tokenRes.status).json({
        error: "CLERK_TOKEN_MINT_FAILED",
        message: text,
      });
    }

    const tok = await tokenRes.json();

    // ── Decode the actual JWT to find its real expiry ──────
    // Clerk silently ignores `expires_in_seconds` on some instances
    // (notably free tier), returning a 60s token instead. Trust the
    // token itself, not our request.
    let actualExp = null;
    try {
      const payloadB64 = tok.jwt.split(".")[1];
      const payloadJson = Buffer.from(payloadB64, "base64url").toString("utf8");
      const payload = JSON.parse(payloadJson);
      actualExp = payload.exp;
    } catch (err) {
      console.error("[dev/token] failed to decode JWT:", err.message);
    }

    const now = Math.floor(Date.now() / 1000);
    const actualTtl = actualExp ? actualExp - now : 60;

    console.log(
      `[dev/token] minted for ${userId} — Clerk issued ${actualTtl}s TTL`
    );

    res.json({
      token: tok.jwt,
      sessionId: session.id,
      expiresIn: actualTtl,
      expiresAt: actualExp,
    });
  } catch (err) {
    console.error("[dev/token] error:", err);
    next(err);
  }
});

export const messagesRouter = express.Router();

// ── Schemas ──────────────────────────────────────────
const attachmentSchema = z.object({
  kind: z.enum(["image", "file", "audio"]),
  url: z.string().url(),
  thumbUrl: z.string().url().nullable().optional(),
  mimeType: z.string().min(3),

  body: z.string().max(4000).default(""),
  attachments: z.array(attachmentSchema).max(10).default([]),
  replyToId: z.string().nullable().optional(),

});

// ── Helpers ──────────────────────────────────────────
function serializeMessage(m, senderUser) {
  return {
    id: String(m._id),
    conversationId: String(m.conversationId),
    senderId: String(m.senderId),
    clientMessageId: m.clientMessageId,
    body: m.body,
    attachments: m.attachments ?? [],
    replyToId: m.replyToId ? String(m.replyToId) : null,
    reactions: m.reactions ?? {},
    editedAt: m.editedAt,
    deletedAt: m.deletedAt,
    localTimestamp: m.localTimestamp,
    createdAt: m.createdAt,
    updatedAt: m.updatedAt,
    sender: senderUser ? serializeUser(senderUser) : null,
  };
}

async function ensureParticipant(conversationId, userId) {
  const convo = await Conversation.findById(conversationId);
  if (!convo) return { convo: null, ok: false };
  const ok = convo.participants.some(
    (p) => String(p.userId) === String(userId) && !p.leftAt
  );
  return { convo, ok };
}

function previewFromBody(body, attachments) {
  if (body && body.trim().length > 0) {
    return body.length > 80 ? body.slice(0, 80) + "…" : body;
  }
  if (attachments.length > 0) {
    const kinds = new Set(attachments.map((a) => a.kind));
    if (kinds.has("image")) return "📷 Image";
    if (kinds.has("audio")) return "🎵 Audio";
    return "📎 File";
  }
  return "";
}

// ── POST /conversations/:id/messages ─────────────────
// Send a message. Idempotent via (senderId, clientMessageId).
messagesRouter.post(
  "/:id/messages",
  requireAuth,
  messageLimiter,
  validate(sendMessageSchema),
  async (req, res, next) => {
    try {
      const { id: conversationId } = req.params;
      if (!mongoose.Types.ObjectId.isValid(conversationId)) {
        return res.status(400).json({ error: "INVALID_ID" });
      }

      const { convo, ok } = await ensureParticipant(conversationId, req.userId);
      if (!convo) {
        return res.status(404).json({ error: "CONVERSATION_NOT_FOUND" });
      }
      if (!ok) {
        return res.status(403).json({ error: "FORBIDDEN" });
      }

      const {
        clientMessageId,
        body,
        attachments,
        replyToId,
        localTimestamp,
      } = req.body;

      // Idempotency — same sender, same clientMessageId.
      const existing = await Message.findOne({
        senderId: req.userId,
        clientMessageId,
      });
      if (existing) {
        const senderUser = await User.findById(existing.senderId).lean();
        return res.status(200).json({
          message: serializeMessage(existing, senderUser),
          duplicate: true,
        });
      }

      // Validate replyToId if present.
      let replyRef = null;
      if (replyToId) {
        if (!mongoose.Types.ObjectId.isValid(replyToId)) {
          return res.status(400).json({ error: "INVALID_REPLY_TO" });
        }
        replyRef = await Message.exists({
          _id: replyToId,
          conversationId,
        });
        if (!replyRef) {
          return res.status(400).json({ error: "REPLY_MESSAGE_NOT_FOUND" });
        }
      }

      const now = new Date();
      const localTs = localTimestamp ? new Date(localTimestamp) : now;

      const msg = await Message.create({
        conversationId,
        senderId: req.userId,
        clientMessageId,
        body: body ?? "",
        attachments: attachments ?? [],
        replyToId: replyToId ?? null,
        localTimestamp: localTs,
      });

      // Bump conversation denormalized fields.
      convo.lastMessageAt = msg.createdAt;
      convo.lastMessagePreview = previewFromBody(msg.body, msg.attachments);
      convo.lastMessageSenderId = msg.senderId;
      convo.messageCount = (convo.messageCount ?? 0) + 1;
      await convo.save();

      // SyncLog for idempotency audit + later sync-batch endpoint.
      await SyncLog.updateOne(
        { userId: req.userId, clientActionId: clientMessageId },
        {
          $setOnInsert: {
            userId: req.userId,
            challengeId: convo.challengeId ?? convo._id, // reuse field
            clientActionId: clientMessageId,
            actionType: "MESSAGE_SEND",
            status: "ACCEPTED",
          },
        },
        { upsert: true }
      );

      const senderUser = await User.findById(req.userId).lean();
      const payload = serializeMessage(msg, senderUser);

      // Real-time broadcast to conversation room.
      emitToConversation(conversationId, "message:new", payload);

      res.status(201).json({ message: payload });
    } catch (err) {
      next(err);
    }
  }
);

// Cursor-paginated. `before` = messageId to fetch older-than.
messagesRouter.get(
  "/:id/messages",
  requireAuth,
  async (req, res, next) => {
    try {
      const { id: conversationId } = req.params;
      if (!mongoose.Types.ObjectId.isValid(conversationId)) {
        return res.status(400).json({ error: "INVALID_ID" });
      }

      const { ok } = await ensureParticipant(conversationId, req.userId);
      if (!ok) {
        return res.status(403).json({ error: "FORBIDDEN" });
      }

      const limit = Math.min(Number(req.query.limit) || 50, 100);
      const before = req.query.before;

      const query = {
        conversationId,
        deletedAt: null,
      };

      if (before) {
        if (!mongoose.Types.ObjectId.isValid(before)) {
          return res.status(400).json({ error: "INVALID_CURSOR" });
        }
        const cursor = await Message.findById(before, { createdAt: 1 }).lean();
        if (cursor) {
          query.createdAt = { $lt: cursor.createdAt };
        }
      }

      const messages = await Message.find(query)
        .sort({ createdAt: -1 })
        .limit(limit)
        .lean();

      // Reverse so response is oldest-first within the page.
      const ordered = messages.reverse();

      // Batch-fetch senders.
      const senderIds = Array.from(new Set(ordered.map((m) => String(m.senderId))));
      const senders = await User.find({ _id: { $in: senderIds } }).lean();
      const senderMap = new Map(senders.map((u) => [String(u._id), u]));

      res.json({
        messages: ordered.map((m) =>
          serializeMessage(m, senderMap.get(String(m.senderId)))
        ),
        hasMore: messages.length === limit,
      });
    } catch (err) {
      next(err);
    }
  }
);

function findDay(challenge, dayId) {
  for (const phase of challenge.phases ?? []) {
    for (const day of phase.days ?? []) {
      if (String(day._id) === String(dayId)) {
        return { day, phase };
      }
    }
  }
  return null;
}

function isLate(day, completedAt, graceHours) {
  const scheduled = new Date(day.scheduledDate).getTime();
  const completed = new Date(completedAt).getTime();
  const graceMs = graceHours * 60 * 60 * 1000;
  return completed > scheduled + graceMs;
}

  return Participant.findOne({ challengeId, userId });

        : new Date();

        found.day,

      );

function previewFromBody(body, attachments) {
  if (body && body.trim().length > 0) {
    return body.length > 80 ? body.slice(0, 80) + "…" : body;
  }
  if (attachments && attachments.length > 0) {
    const kinds = new Set(attachments.map((a) => a.kind));
    if (kinds.has("image")) return "📷 Image";
    if (kinds.has("audio")) return "🎵 Audio";
    return "📎 File";
  }
  return "";
}

function serializeMessage(m) {
  return {
    id: String(m._id),
    conversationId: String(m.conversationId),
    senderId: String(m.senderId),
    clientMessageId: m.clientMessageId,
    body: m.body,
    attachments: m.attachments ?? [],
    replyToId: m.replyToId ? String(m.replyToId) : null,
    reactions: m.reactions ?? {},
    editedAt: m.editedAt,
    deletedAt: m.deletedAt,
    localTimestamp: m.localTimestamp,
    createdAt: m.createdAt,
    updatedAt: m.updatedAt,
  };
}

function isLate(day, completedAt, graceHours) {
  const scheduled = new Date(day.scheduledDate).getTime();
  const completed = new Date(completedAt).getTime();
  const graceMs = graceHours * 60 * 60 * 1000;
  return completed > scheduled + graceMs;
}

function findDay(challenge, dayId) {
  for (const phase of challenge.phases ?? []) {
    for (const day of phase.days ?? []) {
      if (String(day._id) === String(dayId)) {
        return { day, phase };
      }
    }
  }
  return null;
}

async function handleMessageSend(userId, payload) {
  const {
    conversationId,
    clientMessageId,
    body,
    attachments,
    replyToId,
    localTimestamp,
  } = payload;

  if (!mongoose.Types.ObjectId.isValid(conversationId)) {
    return { status: "REJECTED", reason: "INVALID_CONVERSATION_ID" };
  }

  const convo = await Conversation.findById(conversationId);
  if (!convo) {
    return { status: "REJECTED", reason: "CONVERSATION_NOT_FOUND" };
  }

  const isParticipant = convo.participants.some(
    (p) => String(p.userId) === String(userId) && !p.leftAt
  );
  if (!isParticipant) {
    return { status: "REJECTED", reason: "NOT_A_PARTICIPANT" };
  }

  // Idempotency: same sender + clientMessageId
  const existing = await Message.findOne({
    senderId: userId,
    clientMessageId,
  });
  if (existing) {
    return {
      status: "DUPLICATE",
      result: { message: serializeMessage(existing) },
    };
  }

  const now = new Date();
  const localTs = localTimestamp ? new Date(localTimestamp) : now;

  const msg = await Message.create({
    conversationId,
    senderId: userId,
    clientMessageId,
    body: body ?? "",
    attachments: attachments ?? [],
    replyToId: replyToId ?? null,
    localTimestamp: localTs,
  });

  convo.lastMessageAt = msg.createdAt;
  convo.lastMessagePreview = previewFromBody(msg.body, msg.attachments);
  convo.lastMessageSenderId = msg.senderId;
  convo.messageCount = (convo.messageCount ?? 0) + 1;
  await convo.save();

  const payloadOut = serializeMessage(msg);
  emitToConversation(String(conversationId), "message:new", payloadOut);

  return { status: "ACCEPTED", result: { message: payloadOut } };
}

async function handleMessageEdit(userId, payload) {
  const { conversationId, messageId, body } = payload;

  if (
    !mongoose.Types.ObjectId.isValid(conversationId) ||
    !mongoose.Types.ObjectId.isValid(messageId)
  ) {
    return { status: "REJECTED", reason: "INVALID_ID" };
  }

  const msg = await Message.findOne({ _id: messageId, conversationId });
  if (!msg) return { status: "REJECTED", reason: "MESSAGE_NOT_FOUND" };
  if (String(msg.senderId) !== String(userId)) {
    return { status: "REJECTED", reason: "CANNOT_EDIT_OTHERS" };
  }
  if (msg.deletedAt) return { status: "REJECTED", reason: "MESSAGE_DELETED" };

  msg.body = body;
  msg.editedAt = new Date();
  await msg.save();

  const out = serializeMessage(msg);
  emitToConversation(String(conversationId), "message:edited", out);
  return { status: "ACCEPTED", result: { message: out } };
}

async function handleDayComplete(userId, payload) {
  const { challengeSlug, dayId, completedAt, note, artifactUrl } = payload;

  const challenge = await Challenge.findOne({ slug: challengeSlug });
  if (!challenge) return { status: "REJECTED", reason: "CHALLENGE_NOT_FOUND" };

  const participant = await Participant.findOne({
    challengeId: challenge._id,
    userId,
  });
  if (!participant) return { status: "REJECTED", reason: "NOT_A_PARTICIPANT" };

  const found = findDay(challenge, dayId);
  if (!found) return { status: "REJECTED", reason: "DAY_NOT_FOUND" };

  const ts = completedAt ? new Date(completedAt) : new Date();
  const late = isLate(found.day, ts, challenge.settings?.graceHours ?? 48);

  const progress = await Progress.findOneAndUpdate(
    {
      challengeId: challenge._id,
      userId,
      dayId: found.day._id,
    },
    {
      $set: {
        dayNumber: found.day.dayNumber,
        status: "COMPLETED",
        completedAt: ts,
        localTimestamp: ts,
        late,
        note: note ?? null,
        artifactUrl: artifactUrl ?? null,
        syncedAt: new Date(),
      },
      $setOnInsert: {
        challengeId: challenge._id,
        userId,
        dayId: found.day._id,
      },
    },
    { new: true, upsert: true }
  );

  const completions = await Progress.find({
    challengeId: challenge._id,
    userId,
    status: "COMPLETED",
  })
    .select("completedAt")
    .lean();

  const nextStreak = computeStreak(completions, participant.streak);
  participant.streak = nextStreak;
  participant.stats = {
    completed: completions.length,
    missed: Math.max(0, challenge.availableDays - completions.length),
  };
  participant.lastSyncedAt = new Date();
  await participant.save();

  return {
    status: "ACCEPTED",
    result: {
      progress: {
        id: String(progress._id),
        dayId: String(progress.dayId),
        dayNumber: progress.dayNumber,
        status: progress.status,
        completedAt: progress.completedAt,
        late: progress.late,
      },
      streak: participant.streak,
      stats: participant.stats,
    },
  };
}

const HANDLERS = {
  MESSAGE_SEND: handleMessageSend,
  MESSAGE_EDIT: handleMessageEdit,
  MESSAGE_DELETE: handleMessageDelete,
  READ_RECEIPT: handleReadReceipt,
  DAY_COMPLETE: handleDayComplete,
};

export const SUPPORTED_ACTIONS = Object.keys(HANDLERS);

export async function dispatchAction(userId, action) {
  const { type, payload } = action;

  if (!type || typeof type !== "string") {
    return { status: "REJECTED", reason: "MISSING_TYPE" };
  }

  const handler = HANDLERS[type];
  if (!handler) {
    return { status: "REJECTED", reason: `UNSUPPORTED_TYPE: ${type}` };
  }

  try {
    return await handler(userId, payload ?? {});
  } catch (err) {
    console.error(`[sync] handler error for ${type}:`, err);
    return { status: "REJECTED", reason: err.message ?? "HANDLER_ERROR" };
  }
}

const ANILIST_ENDPOINT = "https://graphql.anilist.co";

// Simple in-memory cache to keep us well under the 90 req/min limit.
// key = search query (lowercased), value = { at, results }
const searchCache = new Map();
const CACHE_TTL_MS = 5 * 60 * 1000; // 5 minutes
const CACHE_MAX_ENTRIES = 200;

function cacheKey(q) {
  return q.trim().toLowerCase();
}

function pruneCache() {
  if (searchCache.size <= CACHE_MAX_ENTRIES) return;
  const now = Date.now();
  for (const [k, v] of searchCache.entries()) {
    if (now - v.at > CACHE_TTL_MS) searchCache.delete(k);
  }
}

const SEARCH_QUERY = `
query ($search: String, $page: Int, $perPage: Int) {
  Page(page: $page, perPage: $perPage) {
    pageInfo {
      currentPage
      hasNextPage
    }
    characters(search: $search, sort: [FAVOURITES_DESC, SEARCH_MATCH]) {
      id
      name {
        full
        native
        alternative
      }
      image {
        large
        medium
      }
      description(asHtml: false)
      favourites
      media(perPage: 3, sort: POPULARITY_DESC) {
        nodes {
          id
          title {
            romaji
            english
            native
          }
          genres
          isAdult
        }
      }
    }
  }
}
`;

/**
 * Search AniList for characters matching a query.
 * @param {string} query
 * @param {{ page?: number, perPage?: number }} [opts]
 * @returns {Promise<Array<object>>} sanitized character results
 */
export async function searchCharacters(query, { page = 1, perPage = 20 } = {}) {
  const key = cacheKey(query);
  const cached = searchCache.get(key);
  if (cached && Date.now() - cached.at < CACHE_TTL_MS) {
    return cached.results;
  }

  const body = {
    query: SEARCH_QUERY,
    variables: { search: query, page, perPage },
  };

  const res = await fetch(ANILIST_ENDPOINT, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      Accept: "application/json",
    },
    body: JSON.stringify(body),
  });

  if (res.status === 429) {
    const retry = res.headers.get("retry-after") ?? "60";
    throw Object.assign(new Error("AniList rate limited"), {
      code: "ANILIST_RATE_LIMITED",
      status: 429,
      retryAfter: Number(retry),
    });
  }

  if (!res.ok) {
    const text = await res.text().catch(() => "");
    throw Object.assign(new Error(`AniList error ${res.status}`), {
      code: "ANILIST_ERROR",
      status: res.status,
      body: text.slice(0, 200),
    });
  }

  const json = await res.json();

  if (json.errors?.length) {
    const first = json.errors[0];
    throw Object.assign(new Error(first.message ?? "AniList GraphQL error"), {
      code: "ANILIST_GRAPHQL_ERROR",
      status: 400,
    });
  }

  const edges = json.data?.Page?.characters ?? [];

  // Sanitize: strip HTML from description, normalize shape.
  const results = edges.map((c) => {
    const media = c.media?.nodes?.[0] ?? null;
    return {
      externalId: String(c.id),
      name: c.name?.full ?? c.name?.native ?? "Unknown",
      nameNative: c.name?.native ?? null,
      aliases: c.name?.alternative ?? [],
      image: c.image?.large ?? c.image?.medium ?? null,
      thumb: c.image?.medium ?? null,
      description: stripHtml(c.description ?? "").slice(0, 400),
      favourites: c.favourites ?? 0,
      sourceAnime: media
        ? {
            id: String(media.id),
            title:
              media.title?.english ??
              media.title?.romaji ??
              media.title?.native ??
              null,
            genres: media.genres ?? [],
            isAdult: !!media.isAdult,
          }
        : null,
      attribution: "via AniList",
    };
  });

  searchCache.set(key, { at: Date.now(), results });
  pruneCache();
  return results;
}

/**
 * Validate Gemini's output against the challenge spec.
 * Throws a descriptive error if invalid.
 */
export function validateProgression(progression, spec) {
  const { availableDays, dailyMinutes } = spec;

  // ── Shape checks ──────────────────────────────────
  if (!progression || typeof progression !== "object") {
    throw new Error("Progression is not an object");
  }
  if (!Array.isArray(progression.phases) || progression.phases.length === 0) {
    throw new Error("Progression has no phases");
  }

  // ── Collect all days ──────────────────────────────
  const allDays = [];
  for (const phase of progression.phases) {
    if (!phase.title || !Array.isArray(phase.days)) {
      throw new Error(`Phase "${phase.title ?? "(untitled)"}" is malformed`);
    }
    for (const day of phase.days) {
      allDays.push(day);
    }
  }

  // ── Day count must match ──────────────────────────
  if (allDays.length !== availableDays) {
    throw new Error(
      `Expected ${availableDays} days, got ${allDays.length}`
    );
  }

  // ── Day numbers must be 1..N, unique ──────────────
  const seen = new Set();
  for (const day of allDays) {
    if (typeof day.dayNumber !== "number") {
      throw new Error(`Day "${day.title}" missing dayNumber`);
    }
    if (day.dayNumber < 1 || day.dayNumber > availableDays) {
      throw new Error(`Day number ${day.dayNumber} out of range 1..${availableDays}`);
    }
    if (seen.has(day.dayNumber)) {
      throw new Error(`Duplicate day number: ${day.dayNumber}`);
    }
    seen.add(day.dayNumber);
  }
  for (let i = 1; i <= availableDays; i++) {
    if (!seen.has(i)) {
      throw new Error(`Missing day number: ${i}`);
    }
  }

  // ── Last day must be FINAL ────────────────────────
  const lastDay = allDays.find((d) => d.dayNumber === availableDays);
  if (lastDay?.type !== "FINAL") {
    throw new Error(`Last day (${availableDays}) must have type "FINAL"`);
  }

  // ── Estimated minutes sanity ──────────────────────
  const maxMinutes = dailyMinutes * 1.5;
  for (const day of allDays) {
    if (typeof day.estimatedMinutes !== "number" || day.estimatedMinutes <= 0) {
      throw new Error(`Day ${day.dayNumber} has invalid estimatedMinutes`);
    }
    if (day.estimatedMinutes > maxMinutes) {
      throw new Error(
        `Day ${day.dayNumber} exceeds time budget: ${day.estimatedMinutes} > ${maxMinutes}`
      );
    }
  }

  // ── Requirements must be a non-empty array ────────
  for (const day of allDays) {
    if (!Array.isArray(day.requirements) || day.requirements.length === 0) {
      throw new Error(`Day ${day.dayNumber} has no requirements`);
    }
  }

  return true;
}

/**
 * Sort days within each phase by dayNumber, and sort phases by their first day.
 * Fixes out-of-order output from Gemini.
 */
export function normalizeProgression(progression) {
  for (const phase of progression.phases) {
    phase.days.sort((a, b) => a.dayNumber - b.dayNumber);
  }
  progression.phases.sort((a, b) => {
    const aFirst = Math.min(...a.days.map((d) => d.dayNumber));
    const bFirst = Math.min(...b.days.map((d) => d.dayNumber));
    return aFirst - bFirst;
  });
  return progression;
}

  "application/vnd.ms-excel",

export const SIZE_LIMITS = {
  image: 10 * 1024 * 1024, // 10 MB
  file: 25 * 1024 * 1024,  // 25 MB
};

export function isMimeAllowed(kind, mimeType) {
  if (kind === "image") return IMAGE_MIME_ALLOW.has(mimeType);
  if (kind === "file") return FILE_MIME_ALLOW.has(mimeType);
  return false;
}

export function resourceTypeFor(kind) {
  if (kind === "image") return "image";
  if (kind === "file") return "raw";
  throw new Error(`Unsupported attachment kind: ${kind}`);
}

/**
 * Build the params the client must send to Cloudinary alongside the file.
 * The signature covers these params + api_secret, so nothing can be tampered
 * with client-side.
 *
 * @param {object} opts
 * @param {"image"|"file"} opts.kind
 * @param {string} opts.userId
 * @param {string} [opts.preset] — Cloudinary upload preset name
 * @returns {{ uploadUrl: string, params: Record<string,string|number>, resourceType: string }}
 */
export function buildSignedUpload({ kind, userId, preset = "idus_signed" }) {
  const resourceType = resourceTypeFor(kind);
  const timestamp = Math.floor(Date.now() / 1000);

  // Folder structure: one per user, one per resource type.
  // "attachments/<resourceType>/<userId>" keeps things organized.
  const folder = `attachments/${resourceType}/${userId}`;

  const paramsToSign = {
    folder,
    timestamp,
    upload_preset: preset,
  };

  const signature = cloudinary.utils.api_sign_request(
    paramsToSign,
    config.cloudinary.apiSecret
  );

  const uploadUrl = `https://api.cloudinary.com/v1_1/${config.cloudinary.cloudName}/${resourceType}/upload`;

  return {
    uploadUrl,
    resourceType,
    params: {
      ...paramsToSign,
      api_key: config.cloudinary.apiKey,
      signature,
    },
  };
}

/**

export async function fetchResourceMetadata(publicId, kind) {
  const resourceType = resourceTypeFor(kind);

  try {
    const res = await cloudinary.api.resource(publicId, {
      resource_type: resourceType,
    });

    const url = res.secure_url ?? res.url ?? null;

    // For images, generate a thumbnail by inserting a transformation in the URL.
    // For raw files, no thumbnail.
    let thumbUrl = null;
    if (kind === "image" && url) {
      thumbUrl = cloudinary.url(res.public_id, {
        secure: true,
        resource_type: "image",
        transformation: [
          { width: 400, height: 400, crop: "limit" },
          { quality: "auto", fetch_format: "auto" },
        ],
      });
    }

    return {
      url,
      thumbUrl,
      size: res.bytes ?? null,
      width: res.width ?? null,
      height: res.height ?? null,
      format: res.format ?? null,
      mimeType: res.resource_type === "image"
        ? `image/${res.format}`
        : guessMimeFromFormat(res.format),
    };
  } catch (err) {
    // cloudinary throws 404 for unknown public ids
    console.warn("[cloudinary] resource not found:", publicId, err.message);
    return null;
  }
}

function guessMimeFromFormat(format) {
  if (!format) return "application/octet-stream";
  const map = {
    pdf: "application/pdf",
    zip: "application/zip",
    txt: "text/plain",
    csv: "text/csv",
    md: "text/markdown",
  };
  return map[format.toLowerCase()] ?? "application/octet-stream";
}

  for (let attempt = 1; attempt <= 2; attempt++) {
    try {
      console.log(`[gemini] attempt ${attempt}`);
      const raw = await generateChallengeProgression(spec);
      const normalized = normalizeProgression(raw);
      validateProgression(normalized, spec);
      console.log(`[gemini] attempt ${attempt} passed validation`);

        await new Promise((r) => setTimeout(r, 1000));
      }
    }
  }

  throw new Error(`Gemini generation failed after 2 attempts: ${lastError?.message}`);
}

// src/services/presence.js

// In-memory presence store. Single-process only.
// When you scale to multiple API instances, swap this for Redis
// (a SET per user with instance IDs + TTL heartbeats).

const userSockets = new Map(); // userId -> Set<socketId>
const lastSeen = new Map();    // userId -> Date

const ONLINE_GRACE_MS = 3000;  // treat user as online if within this window

export function addSocket(userId, socketId) {
  const id = String(userId);
  if (!userSockets.has(id)) userSockets.set(id, new Set());
  userSockets.get(id).add(socketId);
  lastSeen.set(id, new Date());
}

export function removeSocket(userId, socketId) {
  const id = String(userId);
  const set = userSockets.get(id);
  if (!set) return false;

  set.delete(socketId);
  lastSeen.set(id, new Date());

  if (set.size === 0) {
    userSockets.delete(id);
    return true; // user just went fully offline
  }
  return false; // user still has another socket
}

export function isOnline(userId) {
  const id = String(userId);
  const set = userSockets.get(id);
  if (!set || set.size === 0) return false;
  return true;
}

export function getPresence(userId) {
  const id = String(userId);
  return {
    userId: id,
    online: isOnline(id),
    lastSeen: lastSeen.get(id) ?? null,
  };
}

export function getPresenceBatch(userIds) {
  return userIds.map((id) => getPresence(id));
}

export function getOnlineUserIds() {
  return Array.from(userSockets.keys());
}
export function getSocketsForUser(userId) {
  const set = userSockets.get(String(userId));
  return set ? Array.from(set) : [];
}

import { createClerkClient, verifyToken } from "@clerk/express";
import { config } from "../config.js";
import { User } from "../models/User.js";
import { Conversation } from "../models/Conversation.js";
import {
  addSocket,
  removeSocket,

  getSocketsForUser,
} from "./presence.js";

const clerk = createClerkClient({
  secretKey: config.clerk.secretKey,
  publishableKey: config.clerk.publishableKey,
});

let io = null;

function roomFor(conversationId) {
  return `conversation:${conversationId}`;
}

export function initSocket(httpServer) {
  io = new Server(httpServer, {
    cors: {
      origin: config.env === "production" ? [config.webUrl] : true,
      credentials: true,
    },
    path: "/socket.io",
  });

  // ── Auth handshake ────────────────────────────────
  io.use(async (socket, next) => {
    try {
      const token =
        socket.handshake.auth?.token ??
        socket.handshake.headers?.authorization?.replace(/^Bearer\s+/i, "");

      if (!token) {
        return next(new Error("UNAUTHENTICATED"));
      }

      const claims = await verifyToken(token, {
        secretKey: config.clerk.secretKey,
      });

      const clerkId = claims?.sub;
      if (!clerkId) {
        return next(new Error("UNAUTHENTICATED"));
      }

      const user = await User.findOne({ clerkId });
      if (!user) {
        return next(new Error("USER_NOT_FOUND"));
      }

      socket.data.userId = String(user._id);
      socket.data.clerkId = clerkId;

      next();
    } catch (err) {
      console.error("[socket] auth error:", err.message);
      next(new Error("UNAUTHENTICATED"));
    }
  });

  // ── Connection ────────────────────────────────────
  io.on("connection", (socket) => {
    const userId = socket.data.userId;
    console.log(`[socket] connected user=${userId} sid=${socket.id}`);

    // ── Presence ─────────────────────────────────
    addSocket(userId, socket.id);

    socket.broadcast.emit("presence:update", {
      userId,
      online: true,
      at: new Date(),
    });

    // ── Join conversation rooms ──────────────────
    socket.on("join:conversations", async () => {
      try {
        const conversations = await Conversation.find(
          { "participants.userId": userId, "participants.leftAt": null },
          { _id: 1 }
        ).lean();

        for (const c of conversations) {
          socket.join(roomFor(c._id));
        }

        socket.emit("joined:conversations", { count: conversations.length });
      } catch (err) {
        console.error("[socket] join:conversations error:", err);
      }
    });

    socket.on("join:conversation", async ({ conversationId }) => {
      if (!conversationId) return;
      try {
        const allowed = await Conversation.exists({
          _id: conversationId,
          participants: { $elemMatch: { userId, leftAt: null } },
        });
        if (!allowed) {
          return socket.emit("error", { error: "NOT_A_PARTICIPANT" });
        }
        socket.join(roomFor(conversationId));

        const convo = await Conversation.findById(conversationId).lean();
        if (convo) {
          const otherIds = convo.participants
            .filter((p) => !p.leftAt && String(p.userId) !== String(userId))
            .map((p) => String(p.userId));

          const presenceList = otherIds.map((id) => ({
            userId: id,
            online: isOnline(id),
          }));
          socket.emit("presence:snapshot", { presence: presenceList });
        }
      } catch (err) {
        console.error("[socket] join:conversation error:", err);
      }
    });

    // ── Typing ───────────────────────────────────
    socket.on("typing:start", ({ conversationId }) => {
      if (!conversationId) return;
      socket.to(roomFor(conversationId)).emit("typing:start", {
        conversationId,
        userId,
        at: new Date(),
      });
    });

    socket.on("typing:stop", ({ conversationId }) => {
      if (!conversationId) return;
      socket.to(roomFor(conversationId)).emit("typing:stop", {
        conversationId,
        userId,
        at: new Date(),
      });
    });

    // ── Disconnect ───────────────────────────────
    socket.on("disconnect", (reason) => {
      const wentOffline = removeSocket(userId, socket.id);
      console.log(
        `[socket] disconnected user=${userId} sid=${socket.id} reason=${reason}`
      );

      if (wentOffline) {
        io.emit("presence:update", {
          userId,
          online: false,
          at: new Date(),
        });
      }
    });
  });

  return io;
}

/**
 * Resolve a schedule into an array of actual challenge dates.
 * @param {Date} startDate
 * @param {Date} endDate
 * @param {{ type: "weekdays" | "custom" | "explicit", days?: number[], dates?: string[] }} schedule
 * @returns {Date[]}
 */
export function resolveChallengeDays(startDate, endDate, schedule) {
  const dates = [];
  const start = new Date(startDate);
  const end = new Date(endDate);

  // Normalize to midnight UTC to avoid time-of-day issues
  start.setUTCHours(0, 0, 0, 0);
  end.setUTCHours(0, 0, 0, 0);

  if (start > end) return dates;

  if (schedule.type === "explicit") {
    // User supplied exact dates
    if (!Array.isArray(schedule.dates)) return dates;
    for (const iso of schedule.dates) {
      const d = new Date(iso);
      if (!isNaN(d) && d >= start && d <= end) {
        d.setUTCHours(0, 0, 0, 0);
        dates.push(d);
      }
    }
    // Sort ascending
    dates.sort((a, b) => a - b);
    return dates;
  }

  // weekdays or custom: use weekday filter
  const allowedDays =
    schedule.type === "weekdays"
      ? [1, 2, 3, 4, 5] // Mon–Fri (JS: 0=Sun, 1=Mon, ...)
      : Array.isArray(schedule.days)
        ? schedule.days
        : [];

  if (allowedDays.length === 0) return dates;

  const cursor = new Date(start);
  while (cursor <= end) {
    if (allowedDays.includes(cursor.getUTCDay())) {
      dates.push(new Date(cursor));
    }
    cursor.setUTCDate(cursor.getUTCDate() + 1);
  }

  return dates;
}

export function computeStreak(completedProgress, existingStreak) {
  const current = existingStreak?.current ?? 0;
  const longest = existingStreak?.longest ?? 0;
  const lastCompletedDate = existingStreak?.lastCompletedDate;

  if (completedProgress.length === 0) {
    return { current: 0, longest, lastCompletedDate };
  }

  const sorted = [...completedProgress].sort(
    (a, b) => new Date(a.completedAt) - new Date(b.completedAt)
  );

  let streak = 1;
  let maxStreak = 1;
  let prev = sorted[0].completedAt;

  for (let i = 1; i < sorted.length; i++) {
    const curr = sorted[i].completedAt;
    const days = daysBetween(prev, curr);

    if (days === 1) streak += 1;
    else if (days > 1) streak = 1;

    maxStreak = Math.max(maxStreak, streak);
    prev = curr;
  }

  const daysSinceLast = daysBetween(prev, new Date());
  if (daysSinceLast > 1) streak = 0;

  return {
    current: streak,
    longest: Math.max(longest, maxStreak),
    lastCompletedDate: prev,
  };
}

function daysBetween(a, b) {
  const MS_PER_DAY = 24 * 60 * 60 * 1000;
  const aMid = new Date(a).setUTCHours(0, 0, 0, 0);
  const bMid = new Date(b).setUTCHours(0, 0, 0, 0);
  return Math.round(Math.abs(bMid - aMid) / MS_PER_DAY);
}
